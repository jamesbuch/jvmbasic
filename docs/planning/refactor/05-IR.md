# Intermediate Representation (IR)

## Overview

The IR sits between the typed AST and bytecode generation, providing:
1. A target-independent representation
2. Optimization opportunities
3. Simplified code generation

## IR Design: Stack-Based with SSA Properties

We use a hybrid approach:
- Stack-based operations (like JVM bytecode)
- SSA-like variable naming for optimization
- Basic blocks for control flow

## IR Instruction Set

```java
package com.jvmbasic.ir;

public sealed interface IRInstruction permits
    // Constants
    IConst, LConst, FConst, DConst, SConst, NullConst,

    // Local Variables
    Load, Store,

    // Arithmetic
    IAdd, ISub, IMul, IDiv, IMod, INeg,
    LAdd, LSub, LMul, LDiv, LMod, LNeg,
    FAdd, FSub, FMul, FDiv, FNeg,
    DAdd, DSub, DMul, DDiv, DNeg,

    // Comparisons
    ICmp, LCmp, FCmp, DCmp, ACmp,

    // Type Conversions
    I2L, I2F, I2D,
    L2I, L2F, L2D,
    F2I, F2L, F2D,
    D2I, D2L, D2F,

    // Object Operations
    New, GetField, PutField, GetStatic, PutStatic,
    InvokeVirtual, InvokeStatic, InvokeInterface, InvokeSpecial,
    NewArray, ArrayLoad, ArrayStore, ArrayLength,

    // Control Flow
    Goto, IfTrue, IfFalse, IfICmpEq, IfICmpNe, IfICmpLt, IfICmpGe, IfICmpGt, IfICmpLe,
    IfACmpEq, IfACmpNe, IfNull, IfNonNull,
    Switch, Return, Throw,

    // Stack Manipulation
    Dup, Pop, Swap,

    // Type Checking
    CheckCast, InstanceOf,

    // Synchronization
    MonitorEnter, MonitorExit,

    // Labels and Phi
    Label, Phi {}
```

## Basic Blocks

```java
public class BasicBlock {
    private final String label;
    private final List<IRInstruction> instructions = new ArrayList<>();
    private final List<BasicBlock> predecessors = new ArrayList<>();
    private final List<BasicBlock> successors = new ArrayList<>();

    public void addInstruction(IRInstruction inst) {
        instructions.add(inst);
    }

    public void addSuccessor(BasicBlock block) {
        successors.add(block);
        block.predecessors.add(this);
    }

    public boolean isTerminated() {
        if (instructions.isEmpty()) return false;
        IRInstruction last = instructions.getLast();
        return last instanceof Goto || last instanceof Return ||
               last instanceof Throw || last instanceof IfTrue ||
               last instanceof IfFalse;
    }
}

public class ControlFlowGraph {
    private final List<BasicBlock> blocks = new ArrayList<>();
    private BasicBlock entryBlock;
    private BasicBlock exitBlock;

    public BasicBlock createBlock(String label) {
        BasicBlock block = new BasicBlock(label);
        blocks.add(block);
        return block;
    }
}
```

## IR Builder

```java
public class IRBuilder {
    private final ControlFlowGraph cfg = new ControlFlowGraph();
    private BasicBlock currentBlock;
    private int nextSlot = 0;
    private final Map<String, Integer> variableSlots = new HashMap<>();

    public void visitFunction(FunctionDecl func) {
        currentBlock = cfg.createBlock("entry");
        cfg.setEntryBlock(currentBlock);

        // Allocate slots for parameters
        for (Parameter param : func.parameters()) {
            int slot = allocSlot(param.name());
        }

        // Generate IR for body
        for (Statement stmt : func.body()) {
            visitStatement(stmt);
        }

        // Ensure function ends with return
        if (!currentBlock.isTerminated()) {
            emit(new Return(mapType(func.returnType())));
        }
    }

    private void visitStatement(Statement stmt) {
        switch (stmt) {
            case DimStmt dim -> {
                int slot = allocSlot(dim.name());
                if (dim.initializer().isPresent()) {
                    visitExpression(dim.initializer().get());
                    emit(new Store(slot, mapType(dim.type())));
                }
            }

            case AssignStmt assign -> {
                visitExpression(assign.value());
                if (assign.target() instanceof IdentifierExpr id) {
                    int slot = variableSlots.get(id.name());
                    emit(new Store(slot, mapType(assign.value().type())));
                } else if (assign.target() instanceof MemberAccessExpr mem) {
                    // Handle field assignment
                    visitExpression(mem.object());
                    emit(new Swap());  // value, obj -> obj, value
                    emit(new PutField(
                        getClassName(mem.object().type()),
                        mem.memberName(),
                        mapType(mem.type())
                    ));
                }
            }

            case IfStmt ifStmt -> {
                String thenLabel = newLabel("then");
                String elseLabel = newLabel("else");
                String endLabel = newLabel("endif");

                visitExpression(ifStmt.condition());
                emit(new IfFalse(elseLabel));

                // Then branch
                currentBlock = cfg.createBlock(thenLabel);
                for (Statement s : ifStmt.thenBranch()) {
                    visitStatement(s);
                }
                emit(new Goto(endLabel));

                // Else branch
                currentBlock = cfg.createBlock(elseLabel);
                if (ifStmt.elseBranch().isPresent()) {
                    for (Statement s : ifStmt.elseBranch().get()) {
                        visitStatement(s);
                    }
                }
                emit(new Goto(endLabel));

                // Continue after if
                currentBlock = cfg.createBlock(endLabel);
            }

            case WhileStmt whileStmt -> {
                String condLabel = newLabel("while_cond");
                String bodyLabel = newLabel("while_body");
                String endLabel = newLabel("while_end");

                emit(new Goto(condLabel));

                // Condition
                currentBlock = cfg.createBlock(condLabel);
                visitExpression(whileStmt.condition());
                emit(new IfFalse(endLabel));

                // Body
                currentBlock = cfg.createBlock(bodyLabel);
                for (Statement s : whileStmt.body()) {
                    visitStatement(s);
                }
                emit(new Goto(condLabel));

                // End
                currentBlock = cfg.createBlock(endLabel);
            }

            case ReturnStmt ret -> {
                if (ret.value().isPresent()) {
                    visitExpression(ret.value().get());
                }
                emit(new Return(ret.value()
                    .map(e -> mapType(e.type()))
                    .orElse(null)));
            }

            case ExpressionStmt expr -> {
                visitExpression(expr.expression());
                // Pop result if not void
                if (expr.expression().type() != PrimitiveType.VOID) {
                    emit(new Pop());
                }
            }
        }
    }

    private void visitExpression(Expression expr) {
        switch (expr) {
            case LiteralExpr lit -> {
                switch (lit.value()) {
                    case Integer i -> emit(new IConst(i));
                    case Long l -> emit(new LConst(l));
                    case Float f -> emit(new FConst(f));
                    case Double d -> emit(new DConst(d));
                    case String s -> emit(new SConst(s));
                    case Boolean b -> emit(new IConst(b ? 1 : 0));
                    case null -> emit(new NullConst());
                }
            }

            case IdentifierExpr id -> {
                int slot = variableSlots.get(id.name());
                emit(new Load(slot, mapType(id.type())));
            }

            case BinaryExpr bin -> {
                visitExpression(bin.left());
                visitExpression(bin.right());
                emitBinaryOp(bin.operator(), bin.left().type());
            }

            case CallExpr call -> {
                // Push arguments
                for (Argument arg : call.arguments()) {
                    visitExpression(arg.value());
                }
                // Call
                if (call.callee() instanceof IdentifierExpr id) {
                    FunctionSymbol func = lookupFunction(id.name());
                    emit(new InvokeStatic(
                        className,
                        id.name(),
                        buildMethodType(func)
                    ));
                }
            }

            case MethodCallExpr mc -> {
                visitExpression(mc.object());
                for (Argument arg : mc.arguments()) {
                    visitExpression(arg.value());
                }
                emit(new InvokeVirtual(
                    getClassName(mc.object().type()),
                    mc.methodName(),
                    buildMethodType(mc.methodName(), mc.object().type())
                ));
            }

            case NewExpr newExpr -> {
                emit(new New(getClassName(newExpr.type())));
                emit(new Dup());
                for (Argument arg : newExpr.arguments()) {
                    visitExpression(arg.value());
                }
                emit(new InvokeSpecial(
                    getClassName(newExpr.type()),
                    "<init>",
                    buildConstructorType(newExpr)
                ));
            }
        }
    }

    private void emitBinaryOp(BinaryOp op, Type operandType) {
        boolean isInt = operandType == PrimitiveType.INTEGER;
        boolean isLong = operandType == PrimitiveType.LONG;
        boolean isFloat = operandType == PrimitiveType.FLOAT;
        boolean isDouble = operandType == PrimitiveType.DOUBLE;

        switch (op) {
            case ADD -> {
                if (isInt) emit(new IAdd());
                else if (isLong) emit(new LAdd());
                else if (isFloat) emit(new FAdd());
                else if (isDouble) emit(new DAdd());
            }
            case SUBTRACT -> {
                if (isInt) emit(new ISub());
                else if (isLong) emit(new LSub());
                else if (isFloat) emit(new FSub());
                else if (isDouble) emit(new DSub());
            }
            // ... more operators
        }
    }
}
```
